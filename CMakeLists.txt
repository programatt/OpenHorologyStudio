cmake_minimum_required(VERSION 3.28)
project(OpenHorologyStudio VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)  # the static libraries also go into the Python module

option(OHS_WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)
option(OHS_ASAN "Build with AddressSanitizer and UBSan" OFF)
option(OHS_WITH_MFEM "Build the MFEM cross-check of the static solver when libmfem (with HYPRE and MPI headers) is found" ON)
option(OHS_TSAN "Build with ThreadSanitizer (cannot be combined with OHS_ASAN)" OFF)
option(OHS_COVERAGE "Instrument for gcov/gcovr line coverage (GCC or Clang)" OFF)
option(OHS_CLANG_TIDY "Run clang-tidy on every translation unit while building" OFF)
option(OHS_BUILD_TESTS "Build the test suite" ON)
option(OHS_BUILD_PYTHON "Build the Python extension module (needs nanobind)" OFF)
option(OHS_BUILD_GUI "Build the Qt 6 desktop application ohs-gui (needs Qt 6 Widgets)" OFF)

option(OHS_WITH_CHRONO "Build the Project Chrono adapter for the level-3 simulation (needs Chrono 9; set Chrono_DIR)" OFF)

find_package(OpenCASCADE REQUIRED)
if(OHS_WITH_CHRONO)
  find_package(Chrono REQUIRED)
endif()
find_package(tomlplusplus REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(spdlog REQUIRED)
find_package(CLI11 REQUIRED)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
# HDF5 results (`ohs fea --hdf5`) are optional: without the library the option reports a clear error.
option(OHS_WITH_HDF5 "Write HDF5 result files when libhdf5 is found" ON)
if(OHS_WITH_HDF5)
  find_package(HDF5 COMPONENTS C HL)
endif()
# MFEM is used serially as an independent check of the structural solver (`ohs fea --mfem`).
if(OHS_WITH_MFEM)
  find_path(OHS_MFEM_INCLUDE mfem.hpp PATH_SUFFIXES mfem)
  find_path(OHS_HYPRE_INCLUDE HYPRE_config.h PATH_SUFFIXES hypre)
  find_library(OHS_MFEM_LIB mfem)
  find_package(MPI COMPONENTS CXX)
  if(OHS_MFEM_INCLUDE AND OHS_HYPRE_INCLUDE AND OHS_MFEM_LIB AND MPI_CXX_FOUND)
    set(OHS_MFEM_FOUND ON)
  else()
    message(STATUS "MFEM not found: ohs fea --mfem will report that the build lacks it")
  endif()
endif()

add_library(ohs_options INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ohs_options INTERFACE -Wall -Wextra -Wpedantic)
  if(OHS_WARNINGS_AS_ERRORS)
    target_compile_options(ohs_options INTERFACE -Werror)
  endif()
  if(OHS_ASAN)
    target_compile_options(ohs_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ohs_options INTERFACE -fsanitize=address,undefined)
  endif()
  if(OHS_TSAN)
    if(OHS_ASAN)
      message(FATAL_ERROR "OHS_TSAN and OHS_ASAN cannot be combined")
    endif()
    # GCC 13 reports false array-bounds / stringop-overflow errors inside libstdc++ when TSan instruments -O2 code.
    target_compile_options(ohs_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer -Wno-array-bounds -Wno-stringop-overflow)
    target_link_options(ohs_options INTERFACE -fsanitize=thread)
  endif()
  if(OHS_COVERAGE)
    target_compile_options(ohs_options INTERFACE --coverage -O0 -g)
    target_link_options(ohs_options INTERFACE --coverage)
  endif()
endif()

if(OHS_CLANG_TIDY)
  find_program(OHS_CLANG_TIDY_EXE NAMES clang-tidy clang-tidy-18 clang-tidy-17 REQUIRED)
  set(CMAKE_CXX_CLANG_TIDY "${OHS_CLANG_TIDY_EXE}" "--config-file=${PROJECT_SOURCE_DIR}/.clang-tidy")
endif()

add_subdirectory(src)

if(OHS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

# ---- Packages: cpack -G DEB | TGZ (Linux), DragNDrop (macOS), WIX (Windows) ----------------------------
set(CPACK_PACKAGE_NAME "ohs")
set(CPACK_PACKAGE_VENDOR "Open Horology Studio contributors")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Watch movement design and analysis")
set(CPACK_PACKAGE_CONTACT "Open Horology Studio <noreply@openhorologystudio.invalid>")
set(CPACK_RESOURCE_FILE_LICENSE "${PROJECT_SOURCE_DIR}/LICENSE.md")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SECTION "science")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_GENERATOR "TGZ")
include(CPack)
